~~~~tsx
import { Component, lazy, Suspense, useEffect, type ReactNode } from "react";
import { useStore } from "./store";
import { TEXT } from "./config";
import { TitleScreen } from "./ui/TitleScreen";
import { Dedication } from "./ui/Dedication";
import { Viewer } from "./ui/Viewer";
import { Hud } from "./ui/Hud";
import { PauseMenu } from "./ui/PauseMenu";
import { Toasts } from "./ui/Toasts";
import { Intro } from "./ui/Intro";
import { Finale } from "./ui/Finale";
import { LoadingScreen } from "./ui/LoadingScreen";
const Scene = lazy(() => import("./scene/Scene"));

class ForestBoundary extends Component<
  { children: ReactNode },
  { failed: boolean }
> {
  state = { failed: false };
  static getDerivedStateFromError() {
    return { failed: true };
  }
  componentDidCatch(error: Error) {
    console.error("Forest error:", error);
  }
  render() {
    return this.state.failed ? (
      <div className="fallback">
        <h2>{TEXT.unavailable}</h2>
        <p>{TEXT.fallback}</p>
        <button className="primary" onClick={() => location.reload()}>
          {TEXT.retry}
        </button>
      </div>
    ) : (
      this.props.children
    );
  }
}
export default function App() {
  const phase = useStore((s) => s.phase),
    paused = useStore((s) => s.paused),
    dedication = useStore((s) => s.dedicationOpen),
    ready = useStore((s) => s.sceneReady),
    large = useStore((s) => s.settings.largeText),
    reduced = useStore((s) => s.settings.reducedMotion);
  useEffect(() => {
    if (useStore.getState().phase === "loading") {
      const timer = setTimeout(
        () => useStore.setState({ phase: "title" }),
        220,
      );
      return () => clearTimeout(timer);
    }
  }, []);
  useEffect(() => {
    const key = (e: KeyboardEvent) => {
      if (e.key !== "Escape" || e.defaultPrevented) return;
      const s = useStore.getState();
      if (s.phase === "viewer") s.closeMemory();
      else if (s.dedicationOpen) useStore.setState({ dedicationOpen: false });
      else if (["explore", "intro", "hanging"].includes(s.phase))
        useStore.setState({ paused: !s.paused });
    };
    window.addEventListener("keydown", key);
    return () => window.removeEventListener("keydown", key);
  }, []);
  const inForest = !["loading", "title", "dedication"].includes(phase);
  return (
    <div
      className={`app ${large ? "large-text" : ""} ${reduced ? "reduced-motion" : ""}`}
    >
      {phase === "loading" && <LoadingScreen progress={70} />}
      {phase === "title" && <TitleScreen />}
      {phase === "dedication" && <Dedication />}
      {inForest && (
        <>
          <ForestBoundary>
            <Suspense fallback={<LoadingScreen progress={30} />}>
              <Scene />
            </Suspense>
          </ForestBoundary>
          {ready && !dedication && <Hud />}
          {ready && <Toasts />}
          {ready && phase === "intro" && <Intro />}
          {phase === "viewer" && <Viewer />}
          {phase === "finale" && <Finale />}
          {paused && <PauseMenu />}
          {dedication && <Dedication reopened />}
        </>
      )}
    </div>
  );
}
~~~~

~~~~ts
import { create } from "zustand";
import { createJSONStorage, persist } from "zustand/middleware";
import { MEMORIES } from "./config";
import { playSound } from "./audio/bridge";
import { revivalRadiusFor } from "./utils/progress";
import {
  createCompatibleStorage,
  normalizeProgress,
  LEGACY_SAVE_KEY,
  SAVE_KEY,
  SAVE_VERSION,
} from "./persistence";

export type Phase =
  | "loading"
  | "title"
  | "dedication"
  | "intro"
  | "explore"
  | "viewer"
  | "hanging"
  | "finale";
export type Tier = "high" | "medium" | "low";
export type Settings = {
  muted: boolean;
  volume: number;
  quality: "auto" | Tier;
  reducedMotion: boolean;
  largeText: boolean;
  skipIntro: boolean;
};
type PreviewBackup = {
  collected: string[];
  hung: string[];
  introSeen: boolean;
  phase: Phase;
  hanging: string[];
  activeMemory: string | null;
  toast: string | null;
};
type State = {
  phase: Phase;
  collected: string[];
  hung: string[];
  hanging: string[];
  introSeen: boolean;
  settings: Settings;
  tier: Tier;
  paused: boolean;
  dedicationOpen: boolean;
  sceneReady: boolean;
  activeMemory: string | null;
  nearMemory: string | null;
  nearTree: boolean;
  toast: string | null;
  audioFailed: boolean;
  revivalRadius: number;
  previewBackup: PreviewBackup | null;
  previewProgress?: (count: number | null) => void;
  setPhase: (phase: Phase) => void;
  setSettings: (v: Partial<Settings>) => void;
  openMemory: (id: string) => void;
  closeMemory: () => void;
  keepMemory: () => void;
  beginHanging: () => void;
  hangOne: (id: string) => void;
  finishHanging: () => void;
  reset: () => void;
};
const safeStorage = createCompatibleStorage();
const mobile =
  typeof matchMedia !== "undefined" && matchMedia("(pointer: coarse)").matches;
const reduced =
  typeof matchMedia !== "undefined" &&
  matchMedia("(prefers-reduced-motion: reduce)").matches;
const validIds = new Set(MEMORIES.map((m) => m.id));
export const useStore = create<State>()(
  persist(
    (set, get) => ({
      phase: "loading",
      collected: [],
      hung: [],
      hanging: [],
      introSeen: false,
      settings: {
        muted: false,
        volume: 0.65,
        quality: "auto",
        reducedMotion: reduced,
        largeText: false,
        skipIntro: false,
      },
      tier: mobile ? "medium" : "high",
      paused: false,
      dedicationOpen: false,
      sceneReady: false,
      activeMemory: null,
      nearMemory: null,
      nearTree: true,
      toast: null,
      audioFailed: false,
      revivalRadius: 0,
      previewBackup: null,
      ...(import.meta.env.DEV
        ? {
            previewProgress: (count: number | null) => {
              const s = get();
              if (count === null) {
                if (!s.previewBackup) return;
                const backup = s.previewBackup;
                set({
                  ...backup,
                  previewBackup: null,
                  revivalRadius: revivalRadiusFor(backup.hung.length),
                });
                return;
              }
              if (!Number.isFinite(count)) return;
              const ids = MEMORIES.slice(
                0,
                Math.max(0, Math.min(MEMORIES.length, Math.round(count))),
              ).map((m) => m.id);
              const backup = s.previewBackup ?? {
                collected: s.collected,
                hung: s.hung,
                introSeen: s.introSeen,
                phase: s.phase,
                hanging: s.hanging,
                activeMemory: s.activeMemory,
                toast: s.toast,
              };
              set({
                previewBackup: backup,
                collected: ids,
                hung: ids,
                hanging: [],
                activeMemory: null,
                introSeen: true,
                phase: "explore",
                toast: null,
                revivalRadius: revivalRadiusFor(ids.length),
              });
            },
          }
        : {}),
      setPhase: (phase) => set({ phase }),
      setSettings: (v) =>
        set((s) => ({
          settings: { ...s.settings, ...v },
          ...(v.quality && v.quality !== "auto" ? { tier: v.quality } : {}),
        })),
      openMemory: (id) => {
        const s = get();
        if (
          validIds.has(id) &&
          (s.phase === "explore" || s.phase === "finale") &&
          !s.paused
        ) {
          playSound("paper");
          set({ activeMemory: id, phase: "viewer" });
        }
      },
      closeMemory: () => set({ activeMemory: null, phase: "explore" }),
      keepMemory: () => {
        const s = get();
        if (!s.activeMemory) return;
        set({
          collected: [...new Set([...s.collected, s.activeMemory])],
          activeMemory: null,
          phase: "explore",
        });
      },
      beginHanging: () => {
        const s = get();
        const ids = s.collected.filter((id) => !s.hung.includes(id));
        if (ids.length && s.phase === "explore" && s.nearTree)
          set({ hanging: ids, phase: "hanging" });
      },
      hangOne: (id) => {
        const s = get();
        if (
          !validIds.has(id) ||
          !s.collected.includes(id) ||
          s.hung.includes(id)
        )
          return;
        const hung = [...s.hung, id];
        set({ hung, revivalRadius: revivalRadiusFor(hung.length) });
      },
      finishHanging: () =>
        set((s) => ({
          hanging: [],
          phase: s.hung.length === MEMORIES.length ? "finale" : "explore",
        })),
      reset: () => {
        safeStorage.removeItem(LEGACY_SAVE_KEY);
        set({
          collected: [],
          hung: [],
          hanging: [],
          introSeen: false,
          phase: "title",
          paused: false,
          activeMemory: null,
          nearMemory: null,
          toast: null,
          sceneReady: false,
          dedicationOpen: false,
          revivalRadius: 0,
          previewBackup: null,
        });
      },
    }),
    {
      name: SAVE_KEY,
      version: SAVE_VERSION,
      storage: createJSONStorage(() => safeStorage),
      // Simulated inventory stays transient; real progress and changed preferences survive reload.
      partialize: (s) => ({
        collected: s.previewBackup?.collected ?? s.collected,
        hung: s.previewBackup?.hung ?? s.hung,
        introSeen: s.previewBackup?.introSeen ?? s.introSeen,
        settings: s.settings,
      }),
      migrate: (persisted) => normalizeProgress(persisted),
      merge: (persisted, current) => {
        const restored = normalizeProgress(persisted);
        const settings = { ...current.settings, ...restored.settings };
        return {
          ...current,
          ...restored,
          settings,
          tier: settings.quality === "auto" ? current.tier : settings.quality,
          revivalRadius: revivalRadiusFor(restored.hung.length),
        };
      },
    },
  ),
);
~~~~

~~~~tsx
export function Toasts() {
  const toast = useStore((s) => s.toast),
    phase = useStore((s) => s.phase),
    ready = useStore((s) => s.sceneReady),
    paused = useStore((s) => s.paused),
    dedication = useStore((s) => s.dedicationOpen),
    collected = useStore((s) => s.collected.length),
    hung = useStore((s) => s.hung.length);
  const complete = hung === MEMORIES.length;
  const key = complete ? "complete" : `${collected}:${hung}`;
  const completionSeen = useRef(false);
  const [hint, setHint] = useState({ key, index: 0 });
  const active = phase === "explore" && ready && !paused && !dedication;
  const index =
    hint.key === key ? hint.index : complete && completionSeen.current ? 1 : 0;
  const text = progressHint(
    collected,
    hung,
    index,
    matchMedia("(pointer: coarse)").matches,
  );

  useEffect(() => {
    if (!complete) completionSeen.current = false;
    if (hint.key !== key) {
      setHint({ key, index: complete && completionSeen.current ? 1 : 0 });
      return;
    }
    if (!active || toast || !text) return;
    const timer = window.setTimeout(
      () => {
        if (complete) completionSeen.current = true;
        setHint((prior) => ({ ...prior, index: prior.index + 1 }));
      },
      complete ? 4000 : 6500,
    );
    return () => clearTimeout(timer);
  }, [active, complete, key, hint.key, hint.index, toast, text]);

  useEffect(() => {
    if (!toast || !active) return;
    const timer = window.setTimeout(() => {
      if (useStore.getState().toast === toast)
        useStore.setState({ toast: null });
    }, 4000);
    return () => clearTimeout(timer);
  }, [toast, active]);

  const message = toast ?? text;
  return active && message ? (
    <div className="toast" key={message} role="status">
      <BrandLogo />
      <span>{message}</span>
    </div>
  ) : null;
}
~~~~

~~~~ts
import { MEMORIES, TEXT, WORLD } from "../config";

export function revivalRadiusFor(hung: number): number {
  const progress = Math.max(0, Math.min(1, hung / MEMORIES.length));
  return progress === 0 ? 0 : 8 + (WORLD.size * 0.8 - 8) * progress ** 1.65;
}

export function progressHint(
  collected: number,
  hung: number,
  index: number,
  touch: boolean,
): string | null {
  if (hung >= MEMORIES.length) return index === 0 ? TEXT.home : null;
  if (collected > hung) return index === 0 ? TEXT.hints[1] : null;
  if (collected === 0) {
    if (index === 0) return touch ? TEXT.touchWalk : TEXT.walk;
    if (index === 1) return TEXT.hints[0];
    return null;
  }
  return index === 0 ? TEXT.hints[0] : null;
}
~~~~

~~~~ts
import {

  MEMORIES,
  TEXT,

/** Report personal text left to customize; never invent a family's memories. */
export function personalTextPlaceholders(): string[] {
  const fields: string[] = [];
  const visit = (value: unknown, path: string) => {
    if (typeof value === "string") {
      if (/\[[^\]]+\]/.test(value)) fields.push(path);
    } else if (Array.isArray(value)) {
      value.forEach((entry, index) => visit(entry, `${path}[${index}]`));
    } else if (value !== null && typeof value === "object") {
      for (const [key, entry] of Object.entries(value))
        visit(entry, `${path}.${key}`);
    }
  };
  for (const [key, value] of Object.entries({
    DEDICATION_TEXT,
    DEDICATION_MESSAGE,
    FINAL_MESSAGE,
    CONTACT_LINK,
    MEMORIES,
    TEXT,
  }))
    visit(value, key);
  return fields;
}

export function warnPersonalTextPlaceholders(): void {
  const fields = personalTextPlaceholders();
  if (fields.length)
    console.warn(
      `The Danny: personalize the bracketed text in src/config.ts: ${fields.join(", ")}`,
    );
~~~~

~~~~ts
function memoryStorage(entries: Record<string, string> = {}) {
  const values = new Map(Object.entries(entries));
  let writes = 0;
  const browserStorage = {
    getItem: (key: string) => values.get(key) ?? null,
    setItem: (key: string, value: string) => {
      writes++;
      values.set(key, value);
    },
    removeItem: (key: string) => {
      values.delete(key);
    },
  };
  return { values, browserStorage, writes: () => writes };
}

      },

    await page.getByRole("button", { name: "Tap to begin" }).click();
    await page.getByRole("button", { name: "Continue", exact: true }).click();
~~~~

~~~~javascript
      () => window.__DANNY_TEST__?.state().sceneReady,
      {},
      { timeout: 60000 },
    );
    console.log(`Sampling ${tier} for ${seconds} seconds`);
    const result = await page.evaluate(
      async ({ tier, seconds }) => {
        const url = performance
          .getEntriesByType("resource")
          .map((entry) => entry.name)
          .find((name) => name.includes("/@react-three_fiber.js"));
        if (!url) throw new Error("Cannot locate the loaded R3F module");
        const fiber = await import(url);
        const root = fiber._roots.get(document.querySelector(".scene canvas"));
        const gl = root.store.getState().gl;
        const context = gl.getContext();
        const extension = context.getExtension("WEBGL_debug_renderer_info");
        const renderer = extension
          ? context.getParameter(extension.UNMASKED_RENDERER_WEBGL)
          : "unknown";
        const originalReset = gl.info.autoReset;
        gl.info.autoReset = false;
        let frames = 0,
          calls = 0,
          triangles = 0,
          first = 0,
          last = 0;
        const began = performance.now();
        const remove = fiber.addAfterEffect(() => {
          const now = performance.now();
          if (now - began >= 4000) {
            if (!frames) first = now;
            last = now;
            frames++;
            calls += gl.info.render.calls;
            triangles += gl.info.render.triangles;
          }
          gl.info.reset();
        });
        await new Promise((resolve) => setTimeout(resolve, seconds * 1000));
        remove();
        gl.info.autoReset = originalReset;
        if (frames < 2) throw new Error("Too few benchmark frames");
        return {
          tier,
          frames,
          fps: +(((frames - 1) * 1000) / (last - first)).toFixed(2),
          calls: Math.round(calls / frames),
          triangles: Math.round(triangles / frames),
          renderer,
          viewport: "960x640",
          deviceScaleFactor: 1,
          sampleMs: Math.round(performance.now() - began),
          geometries: gl.info.memory.geometries,
          textures: gl.info.memory.textures,
        };
      },
      { tier, seconds },
    );
    const cdp = await page.context().newCDPSession(page);
    const heap = await cdp.send("Runtime.getHeapUsage");
    result.heapUsedBytes = heap.usedSize;
    result.heapTotalBytes = heap.totalSize;
    results.push(result);
    console.log(JSON.stringify(result));
    await page.close();
  }
  if (output)
    await writeFile(output, JSON.stringify(results, null, 2) + "\n", "utf8");
} finally {
  await browser.close();
}
~~~~

## README.md

~~~~markdown
# The Danny

A memorial forest, built with love. A complete, static Vite / React 18 / strict TypeScript experience. Forest illustrations, geometry, textures, music, and sound effects are generated by code. The supplied flower logo is an editable SVG, recolored for the memorial forest and shared with the favicon. Instrument Serif Regular is bundled locally through `@fontsource/instrument-serif`; the app makes no requests to third-party font services. Optional photo requests use the URLs you add yourself.

Typography uses Instrument Serif Regular for the serif text, with upright styling throughout, including the generated picture labels. Small utility labels retain their sans-serif styling. Instrument Serif is distributed under the SIL Open Font License 1.1; see the [original font project](https://github.com/Instrument/instrument-serif) and [bundled license](public/fonts/instrument-serif-LICENSE.txt). Change `--font-serif` in `src/styles.css` to choose another family.

The logo lives in `src/assets/blue-heaven-flower.svg`. Its petal, stamen and anther gradients use sage, emerald, lime, cream and warm gold from the forest palette. Edit those SVG color stops to recolor it; keep the five-petal paths if you want to preserve the provided shape. `BrandLogo` reuses the same asset across the UI, and `src/main.tsx` uses it for the favicon. Decorative logos stay hidden from screen readers beside the existing accessible names and labels.

## Run

Requires Node.js 20.19+ or 22.12+.

```sh
npm install
npm run dev
npm run build
npm run preview
```

The dev server listens on port 5173 by default and selects the next available port when occupied; the v2.5 checks use 5174. Open the printed network URL on your phone while connected to the same local network. The production output is `dist/`; no server, database, API keys, or environment variables are required.

## How to personalize

Edit **`src/config.ts`**:

| Field | What to change |
| --- | --- |
| `PROJECT_NAME` | The project title, currently The Danny. |
| `FATHER_FULL_NAME`, `FATHER_NICKNAME` | Danilo Cambronero and Danny. Dedication, inscriptions and friendly hints derive from these values. `TRIBUTE_NAME` inserts the nickname after the first name; edit that expression if your preferred name order differs. |
| `BIRTH_YEAR`, `PASSING_YEAR` | 1964 and 2026. `LIFE_YEARS` uses an en dash with nonbreaking spaces; the dedication date also has a nowrap span. |
| `DEDICATION_TEXT` | The exact dedication sentence, if you want to change its wording. |
| `DEDICATION_MESSAGE` | Replace `[ADD A SHORT, WARM MESSAGE HERE]`. |
| `FINAL_MESSAGE` | Replace `[Thank you for remembering him.]`. |
| `CONTACT_LINK` | Replace `[EMAIL OR SOCIAL LINK]`; the About/Contact tab arrives in Stage F. |
| `AMBIENT_TRACK_URL` | Optional owner-authorized recording URL. Empty uses generated audio; streaming, ducking and source selection arrive in Stage G. |
| `MEMORIES` | All eight `title`, `caption`, `story`, `zone`, and `[x, z]` `position` values. Use `\n` for additional story paragraphs. Keep the eight stable `id` values so existing saved progress continues to match. |
| `MEMORIES[n].imageUrl` | Add an optional HTTPS photo URL. The forest uses a square cover crop inside its generated cream frame; failed loads fall back to the original illustration. |
| `TEXT` | Menu labels, hints, captions, accessible labels, and other interface copy. |
| `PALETTE`, `WORLD`, `QUALITY` | Art colors, world seed / size, and quality settings. |

The defaults intentionally retain the requested clearly marked personal-story placeholders. There are eight finished illustrations: fishing, sunrise, dinner, sailboat, garden, guitar, road trip, and campfire.

Production logs one warning listing config fields containing bracketed placeholders. Replace the family text before publishing; the app does not invent stories to conceal the requested personal-text fields. The dedication sentence appears once, followed by its message. Pause and finale retain their inscriptions.

For photos, put files you own in `public/photos/` and use URLs such as `/photos/dad-fishing.jpg`. Vite copies them to the same origin in production. Alternatively, use a photo host that permits cross-origin access and sends `Access-Control-Allow-Origin: *` (or your site's origin). A publicly viewable URL does not always permit use as a WebGL texture. Same-origin hosting avoids that problem. Keep images compressed, ideally below 1 MB. The app itself ships without any photo files.

Moving a memory also clears a small area around it during seeded world generation. Choose positions on land, inside approximately `[-85, 85]` on both axes. The pond occupies a circle centered at `[19, 5]` with radius `5.7`. Keep positions outside that circle and avoid placing multiple memories on top of each other.

## Controls and journey

Press and hold the forest to walk toward the ground under your pointer. Mouse, touch, and pen share pointer capture. WASD / arrow keys also move the traveler. Release to ease to a stop. Use E / Space or the nearby picture button to read a memory. Keep it, then return to the central tree and hang your carried memories. Kept memories save immediately; hanging memories save one at a time as they land. The tree blooms and the music gains layers with each returned memory.

The title always comes first, followed by the dedication. The first title tap creates and resumes the audio context. The introduction is skippable; returning visitors skip it by default. The heart reopens the dedication. Kept and hung pictures can be reread from their HUD slots, and hung pictures are also tappable on the tree. The finale offers continued exploration.

Pause / Escape opens sound, master volume, quality, reduced motion, large text, intro skipping, and dedication controls. Reset requires a second confirmation. Saved progress and preferences use `the-danny-v2` in localStorage (schema version 2). On first load without a v2 save, the app copies valid `the-danny-v1` data into v2, preserving collected / hung memories, intro status and all six preferences. The v1 backup stays untouched. An existing v2 save always wins, including an empty reset, so old progress cannot return after resetting. Unknown IDs and invalid preferences are ignored; hung IDs are also included in collected IDs. If storage is blocked or its quota is full, progress remains available for the current session. The migration runs before entering the forest.

Reset clears the current v2 progress and intro status while preserving settings, and removes the legacy v1 backup. Migration retains that backup until the player confirms a reset. Neither key transfers to a different site origin.

## Performance notes

| Tier | Shadows | Postprocessing | Grass | Ambient particles | Max DPR |
| --- | --- | --- | --- | --- | --- |
| High | 1024 | Bloom + vignette | 100% | 480 | 1.75 |
| Medium | 512 | Vignette | 60% | 260 | 1.4 |
| Low | Off | Off | 30% | 120 | 1 |

Touch devices start at Medium. Auto uses Drei's PerformanceMonitor to change tiers and stops oscillation with its fallback. A manual choice overrides it. AdaptiveDpr can further reduce pixel density. The subtle paper shader remains in every tier and requires one small fullscreen draw.

Trees, outlines, grass, bushes, flowers, mushrooms, rocks, logs, reeds, lily pads, leaves, blossoms, footprints, butterflies, fireflies, and birds use instance batches. The 200 × 200 world is divided into 25-unit cells; distant chunks are excluded when the player crosses a culling boundary. Visible instances are packed into shared batches, avoiding a draw call for each chunk. Grass additionally shrinks to zero with distance in the vertex shader. All foliage shares one wind material and uniform set. Procedural textures are cached and mipmapped. Unique picture textures and the unique traveler are deliberately separate meshes.

The simulation, camera, shader uniforms, particles, and instance transforms run through refs / mutable objects. React only updates for discrete events such as proximity changes, collecting, landing a memory, or quality changes. Circle collisions use a spatial grid and push outward so movement slides along obstacles. A single tight directional shadow camera follows the player. Frames stop while paused, reading, viewing the dedication, or in a hidden tab; the audio context suspends while hidden.

The title / dedication code is separate from the lazy forest and audio bundles. HUD and toasts wait for scene readiness and are hidden during dedication. Hints reflect carried/returned memories; at 8/8, “Every memory is home.” appears once and dismisses.

Leva, Stats, metrics, and browser-test diagnostics are guarded by `import.meta.env.DEV` and stripped from production. In development, `/?debug=1` exposes wind / light / fog controls, full-frame draw / triangle readouts, Reset progress, Preview progress N / 8 and Restore saved progress. Preview changes the displayed inventory, hung pictures and central-tree growth; a cream ring shows the store's revival radius. Stage C applies that radius to surrounding-tree shaders. Preview never overwrites the original collected/hung/intro save, even after reload; settings changes still persist. ## How the current application works
~~~~

**State and flow.** src/store.ts:15 defines eight typed phases; continued exploration returns to explore. Zustand owns inventory, hanging queues, preferences and discrete UI state. src/scene/runtime.ts:2 holds movement, input and atmosphere outside React. App lazy-loads the 3D scene and audio is imported only after the title gesture. Scene readiness is set in Forest's effect, but App currently mounts HUD before it.

**World, chunking and instancing.** Seed 4217 creates a 200 × 200 world. src/scene/world.ts:42 assigns instances to 25-unit chunks and reserves the clearing, path, pond and memory positions. Flora repacks visible chunks when the traveler crosses a 12.5-unit boundary or changes tier, using a 76-unit chunk-center radius. Eleven kind batches and two canopy outline batches share geometries and one wind material. Grass scales toward zero at distance in the vertex shader. CollisionGrid uses 8-unit spatial cells and circular obstacles. There are generic pine and faceted crown trees; species LOD sets do not exist.

**Quality.** High: 1024 shadows, bloom/vignette, full grass, 480 dust instances, max DPR 1.75. Medium: 512 shadows, vignette, 60% grass, 260 dust, max DPR 1.4; coarse-pointer devices start here. Low: no shadows/composer, 30% grass, 120 dust, DPR 1. A fullscreen grain shader still draws at Low. Butterflies/fireflies and birds retain fixed counts. Drei PerformanceMonitor adapts Auto; manual quality is respected.

**Audio.** TitleScreen creates/resumes an AudioContext in the original gesture, then imports Tone. A singleton owns music/nature/effects buses, master gain and limiter; bridge.ts dispatches effects. Music is an original seeded D-pentatonic arrangement at 54 BPM with default 4/4 meter, pad, FM plucks and bass. Hung progress increases arrangement and viewer ducks it. Nature synthesizes pink-noise wind, brown-noise water, generic bird chirps, dusk insects and proximity chimes. Hidden tabs suspend the context. Pause only pauses transport; continuous wind remains audible.

**Viewer and inventory.** The DOM viewer uses a dialog focus trap, Escape, blurred backdrop, tilted generated polaroid and paragraph fades. Optional photos cover-fit and fall back on errors. Keep saves immediately and animates a cloned picture toward the inventory before it is removed; a store subscription triggers the traveler's hop. Slots reread kept/hung pictures. Dialog cleanup returns to the prior focused element, which is not always the canvas.

**Hanging and tree.** Timeline sequences each carried ID for 2.25 seconds, lands it at 1.65 seconds and saves each landing. Eight stable anchors drive the eased arc. Pre-positioned instance groups grow leaves, blossoms and ground flowers. Picture strings are plain cylinders. All eight lands trigger finale; the camera orbits, music swells, the tribute appears and Keep exploring returns to explore.

**Persistence.** The existing implementation already uses the-danny-v2, schema 2. When absent, a valid v1 envelope is normalized and copied immediately, preserving IDs, all six current preferences and introSeen; the v1 backup stays until confirmed reset. Existing v2 takes precedence, including an empty reset. Hung IDs are unioned into collected IDs. Storage exceptions fall back to session memory; identical writes are suppressed. No migration rewrite or new key is needed. Language, species discovery and independent mix settings do not exist yet.
